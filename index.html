    <script>
        // ==========================================
        // CONFIGURAÇÃO DO FIREBASE (COLE SEUS DADOS AQUI)
        // ==========================================
        const FIREBASE_URL = "https://SEU_PROJETO.firebaseio.com/keys.json"; 
        const ADMIN_PASSWORD = "552904";

        // Variáveis de Controle
        let currentUser = localStorage.getItem('cinder_user');
        let isAdmin = localStorage.getItem('cinder_admin') === 'true';

        // Inicialização
        window.onload = () => {
            if (currentUser) {
                document.getElementById('userNameDisplay').innerText = currentUser;
                if (isAdmin) showView('adminView');
                else showView('userView');
            } else {
                showView('registerView');
            }
        };

        function showToast(msg) {
            const t = document.getElementById('toast');
            t.innerText = msg;
            t.classList.add('show');
            setTimeout(() => t.classList.remove('show'), 3000);
        }

        function showView(viewId) {
            ['registerView', 'loginView', 'adminView', 'userView'].forEach(v => document.getElementById(v).classList.add('hidden'));
            document.getElementById(viewId).classList.remove('hidden');
        }

        function registerUser() {
            const name = document.getElementById('usernameInput').value.trim();
            if (!name) return showToast("Digite um usuário válido.");
            
            localStorage.setItem('cinder_user', name);
            currentUser = name;
            
            if (name.toLowerCase() === 'criador' || name.toLowerCase() === 'admin' || !localStorage.getItem('first_user_registered')) {
                localStorage.setItem('first_user_registered', 'true');
                showView('loginView');
            } else {
                document.getElementById('userNameDisplay').innerText = currentUser;
                showView('userView');
            }
        }

        function loginAdmin() {
            const pass = document.getElementById('adminPassInput').value;
            if (pass === ADMIN_PASSWORD) {
                isAdmin = true;
                localStorage.setItem('cinder_admin', 'true');
                showView('adminView');
                loadKeys();
            } else {
                showToast("Senha incorreta!");
            }
        }

        function skipAdmin() {
            showView('userView');
        }

        function logout() {
            localStorage.removeItem('cinder_user');
            localStorage.removeItem('cinder_admin');
            location.reload();
        }

        // ==========================================
        // LÓGICA DO FIREBASE (KEYS)
        // ==========================================
        async function generateKey() {
            const duration = parseInt(document.getElementById('keyDuration').value);
            const key = 'CINDER-' + Math.random().toString(36).substring(2, 10).toUpperCase();

            const payload = {
                key: key,
                duration: duration,
                used: false,
                created_at: Date.now()
            };

            try {
                // Envia pro Firebase
                const res = await fetch(FIREBASE_URL, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                
                if (res.ok) {
                    showToast(`Key Gerada: ${key}`);
                    loadKeys();
                }
            } catch (err) {
                showToast("Erro ao conectar. Configure o Firebase!");
            }
        }

        async function loadKeys() {
            try {
                const res = await fetch(FIREBASE_URL);
                const data = await res.json();
                const container = document.getElementById('keyListContainer');
                container.innerHTML = '';

                if (!data) {
                    container.innerHTML = '<p style="color: #666; text-align: center;">Nenhuma key gerada.</p>';
                    return;
                }

                for (let id in data) {
                    const item = data[id];
                    const div = document.createElement('div');
                    div.className = `key-item ${item.used ? 'used' : ''}`;
                    
                    const durText = item.duration === 99999999 ? 'Permanente' : (item.duration / 3600) + 'h';
                    div.innerHTML = `
                        <strong>${item.key}</strong> - ${durText} <br>
                        Status: ${item.used ? '🔴 Usada' : '🟢 Disponível'}
                    `;
                    container.appendChild(div);
                }
            } catch (err) {
                console.log("Erro ao carregar keys.");
            }
        }
    </script>
